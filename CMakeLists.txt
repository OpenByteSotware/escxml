cmake_minimum_required(VERSION 3.15)
project(escxml_embedded_runtime VERSION 0.11.0 LANGUAGES C)
add_library(escxml_runtime src/AssertHandler.c src/StateMachine.c src/StateMachineEventQueue.c)
add_library(ESCXML::Runtime ALIAS escxml_runtime)
target_include_directories(escxml_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
set_target_properties(escxml_runtime PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
 target_compile_options(escxml_runtime PRIVATE /W4)
elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(escxml_runtime PRIVATE -Wall -Wextra -Wpedantic)
elseif(CMAKE_C_COMPILER_ID STREQUAL "GHS")
 message(STATUS "Green Hills compiler detected; use project-specific MULTI options")
endif()
option(ESCX_BUILD_TESTS "Build runtime tests" ON)
if(ESCX_BUILD_TESTS)
 enable_testing()
 add_executable(escxml_stage7_test tests/test_stage7.c)
 target_link_libraries(escxml_stage7_test PRIVATE ESCXML::Runtime)
 set_target_properties(escxml_stage7_test PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
 add_test(NAME escxml_stage7_test COMMAND escxml_stage7_test)
endif()
